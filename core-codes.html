    def _init_weights(self):
        """Initialize the LatentKAN vector-field parameters with non-zero values."""
        for name, param in self.kan.named_parameters():
            if param.requires_grad:
                # Use a wider uniform initialization to avoid near-zero vector-field values.
                param.data.uniform_(-self.init_scale * 3, self.init_scale * 3)
        print(f'  ⚠️ Increased initialization scale to {self.init_scale * 3:.2f} for reasonable initial du/dt magnitude')

    def forward(self, u):
        """
        Args:
            u: Current state with shape [..., n_state].
        Returns:
            Value of the latent vector field with shape [..., output_dim].
        """
        original_shape = u.shape
        # Flatten all leading dimensions.
        u_flat = u.reshape(-1, original_shape[-1])  # [N, n_state]

        # Evaluate the KAN-parameterized local latent vector field.
        du_dt_raw = self.kan(u_flat)  # [N, output_dim]

        # Use the raw KAN output without additional scaling.
        du_dt_flat = du_dt_raw

        # Restore the original leading dimensions.
        du_dt = du_dt_flat.reshape(*original_shape[:-1], -1)  # [..., output_dim]
        return du_dt

    def __init__(self, n_state=1, use_kan=True, hidden_dim=16, grid_size=5, spline_order=3,
                 use_rk4: bool = False, time_encoding: str = 'raw', time_dim: int = 64,
                 rk4_num_steps: int = 1, debug: bool = False, time_scale_factor: float = 300.0,
                 max_slices: int = 64, spatial_freqs: int = 4, **kwargs):
        super().__init__()
        self.n_state = n_state
        self.debug = debug
        self.rk4_num_steps = rk4_num_steps  # Number of substeps per interval (default: one RK4 step)
        self.max_slices = max_slices  # Maximum number of slices used for normalization
        self.spatial_freqs = 1  # Number of spatial Fourier-frequency bands

        # Time encoder: 'raw' uses time directly; 'sincos' uses sinusoidal encoding.
        if time_encoding == 'raw':
            self.time_encoder = RawTimeEncoding()
            self.time_dim = 1
        else:  # 'sincos'
            self.time_encoder = TemporalEncoding(dim=time_dim)
            self.time_dim = time_dim

        # Shared latent vector-field parameterizer: KAN or MLP
        # (input dimension = state + time + spatial Fourier features).
        # Include spatial and slice information: x, y, and the slice index with Fourier encoding.
        if self.spatial_freqs > 0:
            self.spatial_dim = 3 + 3 * 2 * self.spatial_freqs
        else:
            self.spatial_dim = 3
        input_dim = n_state + self.time_dim + self.spatial_dim
        output_dim = n_state
        if use_kan:
            self.dynamics = KANDynamics(input_dim, hidden_dim, output_dim, grid_size, spline_order,
                                       debug=debug, time_scale_factor=time_scale_factor)
        else:
            self.dynamics = MLPDynamics(input_dim, hidden_dim, output_dim)

        # Select the numerical integrator used to induce the flow map.
        self.use_rk4 = use_rk4
